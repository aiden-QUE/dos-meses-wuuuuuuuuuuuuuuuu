function createFlower() { const svg = document.getElementById("flowerSvg"); svg.innerHTML = ""; const ns = "http://www.w3.org/2000/svg"; // Centro y distribución de los pétalos. const cx = 300; const cy = 285; const petalCount = 10; const rows = 15; // Tonos de borgoña: las palabras forman también // las zonas claras y oscuras de la flor. const colors = [ "#35101e", "#481329", "#612039", "#7b3049", "#97465d", "#b46a7b", "#d19aa5" ]; let wordIndex = 0; let itemIndex = 0; for (let p = 0; p < petalCount; p++) { const angle = p * Math.PI * 2 / petalCount; // Cada pétalo se construye con filas de palabras. for (let row = 0; row < rows; row++) { const u = (row + 0.5) / rows; // Forma alargada y redondeada del pétalo. const width = 13 + 48 * Math.sin(Math.PI * u); const distance = 28 + u * 168; // Más palabras cerca del centro y en la zona ancha. const count = Math.max( 1, Math.floor(width / 9) ); for (let j = 0; j < count; j++) { const side = count === 1 ? 0 : (j / (count - 1) - 0.5) * 2; const localX = side * width; const localY = distance; const x = cx + localX * Math.cos(angle) - localY * Math.sin(angle); const y = cy + localX * Math.sin(angle) + localY * Math.cos(angle); const word = document.createElementNS(ns, "text"); word.setAttribute("x", x); word.setAttribute("y", y); word.setAttribute("class", "flower-word"); // Orientación de las palabras según el pétalo. word.setAttribute( "transform", `rotate(${angle * 180 / Math.PI - 90}, ${x}, ${y})` ); const colorPosition = Math.min( colors.length - 1, Math.floor( (0.35 + 0.65 * Math.sin(Math.PI * u)) * colors.length ) ); word.setAttribute( "fill", colors[colorPosition] ); word.setAttribute( "font-size", 6 + Math.sin(Math.PI * u) * 2 ); word.style.setProperty( "--delay", `${itemIndex * 0.012}s` ); word.textContent = loveWords[wordIndex % loveWords.length]; svg.appendChild(word); wordIndex++; itemIndex++; } } } // Centro oscuro formado también por palabras. // No se añade círculo ni dibujo detrás. for (let ring = 0; ring < 5; ring++) { const radius = 8 + ring * 5; const count = 8 + ring * 4; for (let i = 0; i < count; i++) { const angle = i * Math.PI * 2 / count; const word = document.createElementNS(ns, "text"); word.setAttribute( "x", cx + Math.cos(angle) * radius ); word.setAttribute( "y", cy + Math.sin(angle) * radius ); word.setAttribute("class", "flower-word"); word.setAttribute("fill", colors[ring % 4]); word.setAttribute("font-size", "6"); word.style.setProperty( "--delay", `${itemIndex * 0.012}s` ); word.textContent = loveWords[wordIndex % loveWords.length]; svg.appendChild(word); wordIndex++; itemIndex++; } } }
